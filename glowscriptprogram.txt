def update_position():
    global vX,vY,aX,aY, tau, inertia, dt, v, theta, dtheta, kinetic_energy
    vX+=aX*dt
    vY+=aY*dt
#    r.pos = vec(poX, poY, 0)
    if inertia > 0:
        a = tau / inertia
    else:
        a = 9.81
    v += a * dt
    if max(len(sphere_list), 1) == 0:
        kinetic_energy = 1/2 * m * vec(vX, vY, 0).mag2 + 1/2 * v**2
    else:
        kinetic_energy = 1/2 * max(len(sphere_list), 1) * m * vec(vX, vY, 0).mag2 + 1/2 * inertia * v**2
#    poX+=vX
#    poY+=vY 
    r.pos+=vec(vX,vY,0)*dt
    moveSpheres(vX*dt,vY*dt)
    theta += v * dt
    dtheta = v * dt
    for i in sphere_list:
        i.rotate(origin = r.pos, angle = dtheta, axis = vec(0,0,1))

def ring_dist(x, y):
    return ((r.pos.x-x) **2 + (r.pos.y-y) **2) - r.radius**2

def getslope(n):
    points = slope.slice(n, n+2)
    dy = points[1].pos.y - points[0].pos.y
    dx = points[1].pos.x - points[0].pos.x
    return dy / dx
